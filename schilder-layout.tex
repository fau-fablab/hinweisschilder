% und hinweisschilder-a7.tex nach \documentclass eingebunden, vorher muss
% \Schildformat (A6 oder A7) definiert sein. Die Schilder selbst stehen in
% schilder.tex.
%
% Jedes Schild ist eine Karte im Querformat mit festem Aufbau:
%
%   +--------------------------------------------+
%   | Titel                                      |  Hintergrund in Ampel-
%   | Modell                                     |  farbe bzw. Verlauf
%   | Status                                     |
%   |--------------------------------------------|
%   | Text                            +--------+ |
%   |                                 |   QR   | |  QR-Code immer gleich groß
%   |                                 +--------+ |
%   | Ansprechpartner              Version       |
%   +--------------------------------------------+
%
% Die Karte ist für A6 gezeichnet, für A7 wird sie auf 1/sqrt(2) verkleinert.
% Titel, Status und Text wählen ihre Schriftgröße selbst (in festen Grenzen),
% damit alles in die Felder passt.
%
% Druckbogen: DIN A4 (A6: quer, 2 x 2 Karten; A7: hoch, 2 x 4 Karten),
% dazwischen Schnittlinien für die Schere. Beim Drucken "Tatsächliche Größe"
% (100 %) wählen, nicht "An Seite anpassen".

% ---------------------------------------------------------------------
% Text passend setzen: größte Schrift zwischen max und min, bei der der
% Inhalt in Breite x Höhe passt. Ergebnis in \passendBox.
% \passend{max pt}{min pt}{Breite}{Höhe}{Inhalt}
% ---------------------------------------------------------------------
\newsavebox{\passendBox}
\newcommand{\passend}[5]{%
	\def\passendGroesse{#1}%
	\passendVersuch{#2}{#3}{#4}{#5}}
\newcommand{\passendVersuch}[4]{%
	\pgfmathsetmacro{\passendZeile}{1.2*\passendGroesse}%
	\sbox{\passendBox}{\parbox[t]{\dimexpr#2\relax}{%
		\fontsize{\passendGroesse pt}{\passendZeile pt}\selectfont\RaggedRight
		\setlength{\parskip}{0.4\baselineskip}#4}}%
	\ifdim\dimexpr\ht\passendBox+\dp\passendBox\relax>#3\relax
		\ifdim\passendGroesse pt>#1pt
			\pgfmathsetmacro{\passendGroesse}{max(#1,\passendGroesse-0.5)}%
			\passendVersuch{#1}{#2}{#3}{#4}%
		\fi
	\fi}

% ---------------------------------------------------------------------
% QR-Code mit FabCube in der Mitte, erzeugt von LaTeX (Paket qrcode).
% Fehlerkorrektur H, damit der Code trotz Logo lesbar bleibt.
% ---------------------------------------------------------------------
\newsavebox{\qrbox}
\newcommand{\qrMitCube}[2]{% {Größe}{URL}
	\sbox{\qrbox}{\qrcode[height=#1,level=H]{#2}}%
	\usebox{\qrbox}%
	\llap{\makebox[\wd\qrbox]{\raisebox{\dimexpr(\ht\qrbox-\dp\qrbox-\height+\depth)/2\relax}{%
		\colorbox{white}{\includegraphics[width=0.22\wd\qrbox]{fab_cube.png}}}}}}

% \karte{Titel}{Status}{Text}, Optionen in \schildFarbe usw.
\newcommand{\karte}[3]{%
	% Kopf
	\pgfmathsetmacro{\kopfB}{\kB-2*\kRand}%
	\ifx\schildModell\empty
		\passend{26}{12}{\kopfB mm}{13mm}{\bfseries #1}\sbox{\titelBox}{\usebox{\passendBox}}%
		\sbox{\modellBox}{}%
		\def\statusY{-\kRand-15}%
	\else
		\passend{24}{12}{\kopfB mm}{10mm}{\bfseries #1}\sbox{\titelBox}{\usebox{\passendBox}}%
		\passend{12}{7}{\kopfB mm}{5mm}{\schildModell}\sbox{\modellBox}{\usebox{\passendBox}}%
		\def\statusY{-\kRand-17}%
	\fi
	\passend{16}{9}{\kopfB mm}{12mm}{\bfseries #2}\sbox{\statusBox}{\usebox{\passendBox}}%
	% Mitte: Text, rechts daneben der QR-Kasten (falls vorhanden)
	\pgfmathsetmacro{\mitteOben}{\statusY-15}%
	\pgfmathsetmacro{\mitteH}{\kH+\mitteOben-\kRand-7}%
	\ifx\schildQR\empty
		\pgfmathsetmacro{\textB}{\kB-2*\kRand}%
	\else
		\pgfmathsetmacro{\textB}{\kB-2*\kRand-\qrKasten-4}%
		\passend{7}{4}{\qrKasten mm-3mm}{8mm}{\centering\sffamily \schildQRText}%
		\sbox{\qrTextBox}{\usebox{\passendBox}}%
	\fi
	\passend{15}{6}{\textB mm}{\mitteH mm}{#3}\sbox{\textBox}{\usebox{\passendBox}}%
	% Fuß
	\sbox{\kontaktBox}{\fontsize{8}{9.6}\selectfont Ansprechpartner: \schildKontakt}%
	\pgfmathsetmacro{\kontaktB}{\kB-2*\kRand-58}% Platz für die Version rechts
	\ifdim\wd\kontaktBox>\kontaktB mm
		\sbox{\kontaktBox}{\resizebox{\kontaktB mm}{!}{\usebox{\kontaktBox}}}%
	\fi
	% Stil für Hintergrund und Rahmen, Farbname schon eingesetzt
	\edef\karteStil{\noexpand\tikzset{karte/.style={hintergrund \schildFarbe, rahmen \schildFarbe}}}%
	\karteStil
	\sbox{\karteBox}{%
	\begin{tikzpicture}[x=1mm, y=1mm, inner sep=0pt]
		\path[use as bounding box] (0,0) rectangle (\kB,-\kH);
		\path[karte, line width=1.2pt, rounded corners=3mm]
			(0,0) rectangle (\kB,-\kH);
		% Kopf
		\node[anchor=north west] at (\kRand,-\kRand) {\usebox{\titelBox}};
		\ifx\schildModell\empty\else
			\node[anchor=north west] at (\kRand,-\kRand-11) {\usebox{\modellBox}};
		\fi
		\node[anchor=north west] at (\kRand,\statusY) {\usebox{\statusBox}};
		\draw[black!70, line width=0.6pt] (\kRand,\mitteOben+2) -- (\kB-\kRand,\mitteOben+2);
		% Mitte
		\node[anchor=north west] at (\kRand,\mitteOben) {\usebox{\textBox}};
		\ifx\schildQR\empty\else
			\fill[white, rounded corners=2mm]
				(\kB-\kRand-\qrKasten,\mitteOben) rectangle (\kB-\kRand,\mitteOben-\qrGroesse-14);
			\node[anchor=north] at (\kB-\kRand-\qrKasten/2,\mitteOben-3)
				{\qrMitCube{\qrGroesse mm}{\schildQR}};
			\node[anchor=north] at (\kB-\kRand-\qrKasten/2,\mitteOben-\qrGroesse-4.5)
				{\usebox{\qrTextBox}};
		\fi
		% Fuß
		\node[anchor=base west] at (\kRand,-\kH+\kRand) {\usebox{\kontaktBox}};
		\node[anchor=base east] at (\kB-\kRand,-\kH+\kRand)
			{\fontsize{5}{6}\selectfont\color{black!70}\ttfamily
				github.com/fau-fablab/hinweisschilder\ifdefined\Version, \Version\fi};
	\end{tikzpicture}}%
	}

\newcounter{schild}  % laufende Nummer, für die Box jedes Schilds
\newcounter{feld}    % Feld auf dem aktuellen Bogen, von 0 an

% Sortierung: Die Karten werden erst gesammelt und am Ende gruppiert auf die
% Bögen gelegt, jede Gruppe beginnt auf einem neuen Bogen:
%   1. eigene Einweisung (einweisung=... außer werkstatt-einweisung)
%   2. allgemeine Werkstatt-Einweisung (einweisung=werkstatt-einweisung)
%   3. ohne Einweisung
% Innerhalb einer Gruppe bleibt die Reihenfolge aus schilder.tex.
\newcommand{\gruppeEigene}{}
\newcommand{\gruppeWerkstatt}{}
\newcommand{\gruppeOhne}{}

% Speichert die Karte aus \karteBox und merkt sie in ihrer Gruppe vor.
\newcommand{\karteSammeln}{%
	\stepcounter{schild}%
	\expandafter\newsavebox\csname schildbox\arabic{schild}\endcsname
	\global\setbox\csname schildbox\arabic{schild}\endcsname\hbox{%
		\scalebox{\Kartenfaktor}{\usebox{\karteBox}}}%
	\ifx\schildEinweisung\empty
		\listxadd\gruppeOhne{\arabic{schild}}%
	\else\ifthenelse{\equal{\schildEinweisung}{werkstatt-einweisung}}
		{\listxadd\gruppeWerkstatt{\arabic{schild}}}
		{\listxadd\gruppeEigene{\arabic{schild}}}%
	\fi}

% Legt Karte Nr. #1 in das nächste freie Feld.
\newcommand{\karteAblegen}[1]{%
	% Feldmitte: Spalte von links, Zeile von oben
	\pgfmathtruncatemacro{\feldZeile}{floor(\thefeld/\Spalten)}%
	\pgfmathtruncatemacro{\feldSpalte}{\thefeld-\feldZeile*\Spalten}%
	\edef\schildAblegen{%
		\noexpand\AddToShipoutPictureFG*{\noexpand\AtPageLowerLeft{%
			\noexpand\tikz[overlay]\noexpand\node[inner sep=0pt] at
				(\the\dimexpr\FeldBreite*\feldSpalte+\FeldBreite/2\relax,
				 \the\dimexpr\paperheight-\FeldHoehe*\feldZeile-\FeldHoehe/2\relax)
				{\noexpand\usebox{\expandafter\noexpand\csname schildbox#1\endcsname}};}}}%
	\schildAblegen
	\stepcounter{feld}%
	\ifnum\value{feld}=\numexpr\Spalten*\Zeilen\relax
		\null\newpage
		\setcounter{feld}{0}%
	\fi}
% angefangenen Bogen ausgeben, damit die nächste Gruppe neu beginnt
\newcommand{\bogenAbschliessen}{%
	\ifnum\value{feld}>0 \null\newpage\setcounter{feld}{0}\fi}
\newcommand{\gruppeAblegen}[1]{%
	\forlistloop{\karteAblegen}{#1}%
	\bogenAbschliessen}
\AtEndDocument{%
	\gruppeAblegen{\gruppeEigene}%
	\gruppeAblegen{\gruppeWerkstatt}%
	\gruppeAblegen{\gruppeOhne}}
